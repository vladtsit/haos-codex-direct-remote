#include <tunables/global>

# complain (not enforcing): first-pass profile, unverified against real hardware; the
# chown/setuid/etc. calls in run.sh need capabilities that aren't granted below yet.
# Once a run with no denials shows up in `journalctl _TRANSPORT="audit" -g 'apparmor="ALLOWED"'`,
# remove the `complain` flag to actually enforce this profile.
profile codex_remote flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  file,
  network,
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability sys_chroot,
  capability audit_write,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) set=(kill,term,int,hup,cont),

  # entrypoint and everything it execs: bash, curl, git, ssh-keygen, sshd, su-exec, codex
  /run.sh r,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /sbin/** ix,

  # persistent state (codex identity, ssh host keys, authorized_keys) and project files
  /data/** rw,

  # optional read-only Home Assistant config access (only present if `map: homeassistant_config` is used);
  # actual mount point is /homeassistant, not /homeassistant_config (HAOS mounts it 0700 root:root,
  # so reading it as the unprivileged codex user goes through sudo instead)
  /homeassistant/** r,
  /etc/sudoers.d/** r,

  # sshd (privsep chroot, pty allocation, host key generation)
  /etc/ssh/** rw,
  /var/empty/** r,
  /dev/pts/* rw,
  /dev/ptmx rw,
  /dev/tty rw,
  /dev/null rw,
  /dev/urandom r,
  /proc/sys/kernel/random/uuid r,
  /run/** rwk,
  /tmp/** rwk,
}
